\cvsection{Summary}
M.Sc.\ student in Computer Science at the Federal University of Bahia and
software engineering researcher. Experienced in programming with Rust, C, C++,
Java, TypeScript, Go, Python, and more, with hands-on knowledge of databases,
Linux systems development and maintenance, CI/CD, Kubernetes, service meshes,
computer networks, and internet protocols. Strongly interested in software
security, LLM-based agents, highly configurable systems, software architecture,
and tools that support software development.

\begin{bullets}
  \item Integrates continuous-variable quantum key distribution with classical
        network infrastructure at QuIIN (Quantum Industrial Innovation),
        the EMBRAPII CIMATEC Competence Center in Quantum Technologies.
\end{bullets}
\vspace{5pt}
\entry{\textbf{Security Vulnerabilities in Highly Configurable Systems}, UFBA}{2024--present}
\detail{PI: Eduardo Santana de Almeida. Funded by CNPq/MCTI/FNDCT (Conhecimento Brasil Program).}
\begin{bullets}
  \item Studies how vulnerabilities affect specific configurations of open-source
        highly configurable systems.
  \item Rather than only filtering which patches apply to a configuration, looks for
        alternative configurations that avoid a vulnerability without applying its
        patch, using machine-learning-based configuration sampling and Markov chain
        Monte Carlo (MCMC).
  \item Measures how prevalent these vulnerabilities are in deployed systems using
        Shodan data. Resulted in publications [\ref{pub:fse}], [\ref{pub:msr}], and [\ref{pub:sbes}].
\end{bullets}
\vspace{5pt}
\entry{\textbf{ARGUS: AI for Corporate Security}, UFBA}{2023--2025}
\detail{University--industry R\&D project. PI: Gustavo Bittencourt Figueiredo.}
\begin{bullets}
  \item Context-aware platform for smart environments that processes data from
        security cameras and sensors, orchestrates cloud resources, and recommends
        responses to security events.
  \item Resulted in publications [\ref{pub:icsa}] and [\ref{pub:sbqs}].
\end{bullets}
\vspace{5pt}
\entry{\textbf{Intelligent Platform for Reducing Falls in Hospital Beds}, UFBA}{2023--2025}
\detail{PI: Eduardo Santana de Almeida. Funded by CNPq. Partner: Instituto Cardio Pulmonar da Bahia.}
